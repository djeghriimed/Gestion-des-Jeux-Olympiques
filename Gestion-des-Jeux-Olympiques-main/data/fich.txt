Create Table LesSportifs
(
  numSp NUMBER(4),
  nomSp VARCHAR2(20),
  prenomSp VARCHAR2(20),
  paysSp VARCHAR2(20),
  categorieSp VARCHAR2(10),
  dateNaisSp DATE,
  CONSTRAINT SP_CK1 CHECK(numSp between 1000 and 1500),
  CONSTRAINT SP_CK2 CHECK(categorieSp IN ('feminin','masculin')),
  CONSTRAINT SP_CK3 UNIQUE(nomSp, prenomSp)
);

Create Table LesEquipes

 numEq NUMBER(4) PRIMARY KEY,
 paysEq VARCHAR2(20),
 CONSTRAINT EQ_ck1 CHECK(numEq between 1 and 100)

Create Table EquipeSportif
(
  numSp NUMBER(4),
  numEq NUMBER(4),
  CONSTRAINT ES_PK PRIMARY KEY (numSp, numEq),
  CONSTRAINT ES_FK1 FOREIGN KEY (numSp) REFERENCES LesSportifs(numSp),
  CONSTRAINT ES_FK2 FOREIGN KEY (numEq) REFERENCES LesEquipes(numEq)
);

CREATE TRIGGER Verif_Pays_Equipe_Sportif
BEFORE INSERT ON EquipeSportif
FOR EACH ROW
WHEN (
      (SELECT paysEq FROM LesEquipes WHERE numEq = NEW.numEq)
      <>
      (SELECT paysSp FROM LesSportifs WHERE numSp = NEW.numSp)
)
BEGIN
    SELECT RAISE(ABORT, 'Le pays du sportif doit être le même que celui de l''équipe');
END;

  participant NUMBER(4),
  CONSTRAINT I_PK PRIMARY KEY (numEp, participant),
  CONSTRAINT I_FK1 FOREIGN KEY (numEp) REFERENCES LesEpreuves(numEp)

);

/*trigger aussi pour verif que si fome ep indiv alors particpant doit etre un sportif et si par equipe alors une equipe meme chose pour categorie*/
CREATE TRIGGER verif_participant
BEFORE INSERT ON Inscrits
FOR EACH ROW
WHEN NOT (
     (NEW.participant BETWEEN 1000 AND 1500
      AND EXISTS (SELECT 1 FROM LesSportifs WHERE numSp = NEW.participant))
  OR (NEW.participant BETWEEN 1 AND 100
      AND EXISTS (SELECT 1 FROM LesEquipes WHERE numEq = NEW.participant))
)
BEGIN
    SELECT RAISE(ABORT, 'participant doit être un sportif ou une équipe');
END;
